   name: generate snake animation
   on:
     schedule:
       - cron: "0 \*/6 \* \* \*"
     workflow\_dispatch: {}
     push:
       branches: \[ main ]

   permissions:
     contents: write

   jobs:
     build:
       runs-on: ubuntu-latest

             target\_branch: output
             build\_dir: dist
